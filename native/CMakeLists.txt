cmake_minimum_required(VERSION 3.21)
project(mvo_rtc LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_definitions(MBEDTLS_SSL_DTLS_SRTP)
if(MVO_ZIG_EXECUTABLE)
  set(CMAKE_C_ARCHIVE_CREATE "\"${MVO_ZIG_EXECUTABLE}\" ar qc <TARGET> <OBJECTS>")
  set(CMAKE_C_ARCHIVE_FINISH "\"${MVO_ZIG_EXECUTABLE}\" ranlib <TARGET>")
  set(CMAKE_CXX_ARCHIVE_CREATE "\"${MVO_ZIG_EXECUTABLE}\" ar qc <TARGET> <OBJECTS>")
  set(CMAKE_CXX_ARCHIVE_FINISH "\"${MVO_ZIG_EXECUTABLE}\" ranlib <TARGET>")
endif()
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${MBEDTLS_SOURCE}" mbedtls)
add_library(MbedTLS::MbedTLS ALIAS mbedtls)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_DEPS_LIBS OFF CACHE BOOL "" FORCE)
set(sctp_werror OFF CACHE BOOL "" FORCE)
# Localhost candidates are needed by the real ICE/DTLS/SCTP integration tests.
set(ENABLE_LOCALHOST_ADDRESS ON CACHE BOOL "" FORCE)
add_subdirectory("${DATACHANNEL_SOURCE}" datachannel)
add_library(mvo_rtc SHARED rtc_bridge.c)
if(WIN32)
  set_target_properties(mvo_rtc PROPERTIES PREFIX "")
endif()
target_link_libraries(mvo_rtc PRIVATE datachannel)
if(NOT WIN32)
  find_package(Threads REQUIRED)
  target_link_libraries(mvo_rtc PRIVATE Threads::Threads)
endif()
install(TARGETS mvo_rtc datachannel RUNTIME DESTINATION . LIBRARY DESTINATION .)
